Q) Practice — Find the Error

#include <iostream>
using namespace std
int main() {
cout << "Hello"
return 0;
}

Answer:
There are two errors in the program:

1. A semicolon (;) is missing after "using namespace std".
2. A semicolon (;) is missing after cout << "Hello".

Correct Code:

#include <iostream>
using namespace std;

int main() {
cout << "Hello";
return 0;
}

Output:
Hello

Q) Practical Program: Simple Calculator

#include <iostream>
using namespace std;

int main() {
double num1, num2;
char op;

```
cout << "Enter first number: ";
cin >> num1;

cout << "Enter operator (+, -, *, /): ";
cin >> op;

cout << "Enter second number: ";
cin >> num2;

double result;

if (op == '+') 
    result = num1 + num2;
else if (op == '-') 
    result = num1 - num2;
else if (op == '*') 
    result = num1 * num2;
else if (op == '/') 
    result = num1 / num2;
else {
    cout << "Invalid operator";
    return 0;
}

cout << "Result = " << result << endl;

return 0;
```

}

Sample Input:
10
+
5

Sample Output:
Enter first number: 10
Enter operator (+, -, *, /): +
Enter second number: 5
Result = 15
